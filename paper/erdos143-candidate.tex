\begin{abstract}
We construct a countably infinite set $A\subset\mathbb Q\cap[2,\infty)$
with $|ka-b|\geq1$ for all distinct $a,b\in A$ and all positive integers
$k$, but $\sum_{a\in A}1/(a\log a)=\infty$. This gives a negative answer
to the convergence question in Erd\H{o}s problem 143.
The proof is a finite extension argument: each step preserves all earlier
points and increases the sum by at least a fixed positive constant.
New points are added in blocks, each a rational dilation of a finite
primitive set of integers.
Two primes in each dilation factor serve different roles: one controls
separation from later blocks, while the other makes the displacement
from integers as small as needed. The required primitive sets are selected from
periodic sets by an elementary Euler-product argument. Every constructed
point lies within $m^{-12}/2$ to the right of an integer $m$.
A companion Lean~4 project proves the existence theorem and the
periodic-selection lemma.
\end{abstract}

\section{Introduction}
\label{sec:introduction}
Write $\N=\{1,2,\ldots\}$, and let $\log$ denote the natural logarithm.
For $x>1$ and a finite or countable set $B\subset(1,\infty)$, write
\[
 \wt(x)=\frac1{x\log x},\qquad W(B)=\sum_{x\in B}\wt(x)\in[0,\infty].
\]
The sum is the supremum of its finite subsums. We call $A\subset(1,\infty)$
\emph{dilation-separated} if
\begin{equation}\label{eq:original}
 |ka-b|\geq1\qquad(a,b\in A,\ a\ne b,\ k\in\N).
\end{equation}
Both orders of each pair are included. In particular, taking $k=1$
shows that distinct points are at least one apart. Such a set has only
finitely many points in every bounded interval and is at most countable.

For a set of integers greater than one, \eqref{eq:original} is equivalent
to \emph{primitivity}: no element divides another distinct element.
Erd\H{o}s proved that every primitive set of integers has finite
weight $W$~\cite{Erdos1935}. In problem~II of~\cite[p.~692]{Erdos1948},
he asked whether the analogous separation condition for real numbers
implies both convergence of $W(A)$ and vanishing logarithmic density.
These are the two questions in problem~143 of~\cite{Bloom143}.
Koukoulopoulos, Lamzouri, and Lichtman~\cite[Theorem~1]{KLL2025}
proved that every dilation-separated set satisfies
\begin{equation}\label{eq:logdensity}
 \sum_{a\in A\cap[1,x]}\frac1a=o(\log x).
\end{equation}
We give a counterexample to the convergence assertion. Thus the two
questions have different answers, even when all points are rational
and extremely close to integers.

\begin{theorem}\label{thm:main}
There exists a countably infinite set $A\subset\Q\cap[2,\infty)$ with
no finite accumulation point such that \eqref{eq:original} holds and
\begin{equation}\label{eq:divergence}
 W(A)=\infty.
\end{equation}
Every $a\in A$ can be written as
\begin{equation}\label{eq:near-integer}
 a=m+\eta,
 \qquad m\in\N,\quad 0<\eta\leq\frac12m^{-12}.
\end{equation}
More precisely, the construction produces one nested sequence of finite
dilation-separated sets $F_1\subset F_2\subset\cdots$ whose union is $A$ and for which
\begin{equation}\label{eq:prefix-weight}
 W(F_R)\geq\frac{R}{576}\qquad(R\in\N).
\end{equation}
\end{theorem}

The construction must preserve separation from every integer multiple
of every earlier point while adding an infinite total weight.
Separation within one new finite set is easy. If $P$ is primitive and $c>1$, then
$|ckn-cn'|\geq c$ for distinct $n,n'\in P$ and $k\in\N$.
The difficulty is making many such sets compatible with each other.

We use multipliers of the form
\[
 D\frac{pq}{pq-1}=D+\frac{D}{pq-1},
\]
where $D$ is a positive integer and $p,q$ are primes. The prime $q$ is chosen
last, so it can make every displacement in the new finite set as small
as required. The prime $p$ controls later exclusions: on a suitable
new integer grid, every location not divisible by $p$ stays more than
$1$ away from every integer multiple of every point in this block.
Definition~\ref{def:admissible} records the divisibility conditions
that make this possible.
Section~\ref{sec:construction} proves the finite extension in
Proposition~\ref{prop:finite-extension}; iterating it gives
Theorem~\ref{thm:main}.

The only analytic input to this construction is that a periodic set of
integers of density greater than $2/3$ contains, above any prescribed
threshold, a finite primitive subset of weight greater than $1/6$.
We state this as Lemma~\ref{thm:selection} before the construction and
prove it in Section~\ref{sec:analytic}. The constants and the exponent
$12$ are convenient choices; no optimality is claimed.

\section{The finite extension argument}
\label{sec:construction}
A set $E\subset\N$ is \emph{periodic} if membership depends only on the
residue modulo some positive integer $L$. Its density $\dens(E)$ is the
proportion of allowed residue classes modulo $L$; this proportion is
independent of the chosen period.

\begin{lemma}[Periodic selection]\label{thm:selection}
Let $E\subset\N$ be periodic, with $\dens(E)>2/3$. For every integer
$N\geq2$, there is a finite primitive set
$P\subset E\cap[N,\infty)$ such that $W(P)>1/6$.
\end{lemma}

The proof of this lemma is given in Section~\ref{sec:analytic}. We now
show how it gives the finite extension needed for the theorem.

\subsection{Admissible finite configurations}

A block has an integer $D_i\geq48$, a finite nonempty primitive set
$P_i\subset\N\setminus\{1\}$, and two primes $p_i,q_i$. Its points are
\begin{equation}\label{eq:block-data}
 A_i=\{a_{i,n}:n\in P_i\},\qquad
 a_{i,n}=\frac{D_i n p_iq_i}{p_iq_i-1}
        =D_i n+\frac{D_i n}{p_iq_i-1}.
\end{equation}
Thus a block is a common rational dilation of $D_iP_i$, with a positive
displacement that can be made arbitrarily small by increasing the
auxiliary prime $q_i$. We call $p_i$ the \emph{private prime} of the
block: it does not divide $p_iq_i-1$, and the following condition
prevents it from dividing $q_jD_jn$ in any block, including its own.

\begin{definition}\label{def:admissible}
For a finite ordered list of blocks, write $F=\bigcup_{i=1}^t A_i$.
We call the list \emph{admissible} if the following conditions hold.
\begin{enumerate}
\item[(i)] $F$ is dilation-separated, and every point of $A_i$ exceeds
every point of $A_j$ by more than $1$ whenever $j<i$.
\item[(ii)] The primes $p_i$ are distinct and greater than $2$, and
\begin{equation}\label{eq:private-invariant}
 p_j\nmid q_iD_i n
 \qquad(1\leq i,j\leq t,\ n\in P_i).
\end{equation}
\item[(iii)] For every $i$ and $n\in P_i$,
\begin{equation}\label{eq:block-size}
 \frac{a_{i,n}}{p_i}>4,
 \qquad
 0<a_{i,n}-D_i n\leq\frac12(D_i n)^{-12}.
\end{equation}
\item[(iv)] The private primes obey the strict bound
\begin{equation}\label{eq:reciprocal-primes}
 \sum_{i=1}^t\frac1{p_i}<\frac1{100}.
\end{equation}
\end{enumerate}
The empty list is admissible. Put $M(F)=\max F$ when $F\ne\varnothing$,
and $M(\varnothing)=0$.
\end{definition}

The conditions serve two purposes. Condition (i) records the separation
already achieved. Conditions (ii)--(iv) make further extensions possible:
(ii) allows a common multiple of all the integers $q_iD_i n$ that is
divisible by none of the private primes; the first bound in (iii) then
gives a separation margin, while (iv) keeps the excluded proportion of
integers small. The displacement bound in (iii) records the additional
conclusion of Theorem~\ref{thm:main}. We retain all these arithmetic data
when extending a configuration.

\begin{proposition}[Finite extension]\label{prop:finite-extension}
Every admissible finite configuration with point set $F$ can be extended
by appending finitely many blocks to an admissible configuration with
point set $F'$ such that
\[
 W(F'\setminus F)\geq\frac1{576}.
\]
All preceding block data, and hence every point of $F$, remain unchanged.
\end{proposition}

One application of this proposition will add several blocks with a
common integer $D$; we call this group of insertions a \emph{round}.
The gain from a single block depends on $D$, whereas the gain from a
complete round has the fixed lower bound above. We first deduce the
theorem by iterating complete rounds, then prove that each round can
be carried out.

\begin{proof}[Proof of Theorem~\ref{thm:main}]
Start with the empty configuration and successively choose an extension
given by Proposition~\ref{prop:finite-extension}. If $F_R$ is the point
set after $R$ extensions, then these choices give one nested sequence
\[
 F_0=\varnothing\subset F_1\subset F_2\subset\cdots,
 \qquad W(F_R)\geq\frac R{576}.
\]
Set $A=\bigcup_{R\geq1}F_R$. Every point is rational and greater than
$2$, by~\eqref{eq:block-data}. Given distinct $a,b\in A$, both belong
to some $F_R$, which satisfies $|ka-b|\geq1$ for every $k\in\N$.
Thus $A$ is dilation-separated. The same $R$ works for every multiplier;
no limiting argument in $k$ is involved.

The set $A$ is countable, and its separation for $k=1$ implies local
finiteness. The displacement bound~\eqref{eq:near-integer} follows
from~\eqref{eq:block-size}, since each inserted point is retained with
its original data. Finally,
\[
 W(A)\geq W(F_R)\geq R/576\qquad(R\geq1),
\]
so $W(A)=\infty$ and $A$ is infinite. The sets $F_R$ also
satisfy~\eqref{eq:prefix-weight}.
\end{proof}

\subsection{Choosing the next integer scale}

To prove Proposition~\ref{prop:finite-extension}, we seek new points
near integer locations $D\ell$, where $\ell\in\N$. We call $D\N$
the grid and $\ell$ the label of a location. The following lemma chooses
$D$ so that the preceding blocks exclude only labels divisible by their
private primes. The preceding values $D_i$ and all old points stay fixed.

\Needspace{12\baselineskip}
\begin{lemma}\label{lem:grid}
For every admissible configuration with point set $F_0$, there is an
integer $D\geq48$ such that
\begin{equation}\label{eq:grid-choice}
 q_iD_i n\mid D\quad(n\in P_i),
 \qquad p_i\nmid D\quad(1\leq i\leq t).
\end{equation}
On this grid, if $\ell\in\N$ and $p_i\nmid\ell$, then
\begin{equation}\label{eq:grid-separation}
 |ka_{i,n}-D\ell|>4
 \qquad(n\in P_i,\ k\in\Z).
\end{equation}
If $p_i\mid\ell$, then $D\ell$ is an integer multiple of every point
in $A_i$.
\end{lemma}

\begin{proof}
Take the product of the finitely many integers $q_iD_i n$ and multiply
it by a sufficiently large power of $2$. The resulting integer $D$ is
at least $48$ and is divisible by every $q_iD_i n$. By
\eqref{eq:private-invariant}, and because all $p_i$ are odd, none of the
$p_i$ divides $D$. For the empty configuration take $D=48$.

Fix $a=a_{i,n}$, and write $v=D/(q_iD_i n)$ for the corresponding
integer quotient. Then
\begin{equation}\label{eq:integer-gap}
 |ka-D\ell|
   =\frac a{p_i}\left|kp_i-v(p_iq_i-1)\ell\right|.
\end{equation}
Since $p_i\nmid v$ and $p_iq_i-1\equiv-1\pmod{p_i}$, the integer
inside the absolute value is nonzero when $p_i\nmid\ell$. Its absolute
value is therefore at least $1$, and~\eqref{eq:block-size} gives
\eqref{eq:grid-separation}. When $p_i\mid\ell$, the ratio
$D\ell/a=v(p_iq_i-1)\ell/p_i$ is an integer.
\end{proof}

Consequently, the labels $\ell$ avoiding the multiples of all preceding
private primes have density at least $1-\sum_i p_i^{-1}>99/100$.
For these labels, $D\ell$ stays more than $4$ away from
every integer multiple of an old point. A separate estimate controls
the blocks that will be inserted before this grid is changed again.

\subsection{The density excluded by one block}

Once $D$ has been fixed, we must also avoid multiples of the new blocks
inserted on this grid. All multiples of a block are multiples of the
single rational number $Dpq/(pq-1)$. The following set contains all
residue classes in which a distance of at most $1$ can occur.
For $D\geq1$ and primes $p,q$ with $pq\geq2D$, let
\begin{equation}\label{eq:bad-residues}
 \mathcal B_{p,q}(D)=
 \left\{z\bmod pq:z\in\Z,\quad
 |z|\leq\left\lfloor\frac{pq-1}{D}\right\rfloor\right\}.
\end{equation}
The bound below is independent of the number of points in the block.

\begin{lemma}\label{lem:cost}
If $\ell\in\N$ and $\ell\bmod pq\notin\mathcal B_{p,q}(D)$, then
\[
 \left|\frac{Dpq}{pq-1}h-D\ell\right|>1
 \qquad(h\in\Z).
\]
Moreover,
\begin{equation}\label{eq:cost}
 \frac{|\mathcal B_{p,q}(D)|}{pq}\leq\frac3D.
\end{equation}
\end{lemma}

\begin{proof}
If the displayed absolute value is at most $1$, then the integer
$z=pqh-(pq-1)\ell$ satisfies $D|z|\leq pq-1$. But
$z\equiv\ell\pmod{pq}$, so the residue of $\ell$ belongs to
$\mathcal B_{p,q}(D)$. For the density bound, reduction modulo $pq$
cannot increase cardinality, and hence
\[
 \frac{|\mathcal B_{p,q}(D)|}{pq}
 \leq\frac{2\lfloor(pq-1)/D\rfloor+1}{pq}
 \leq\frac2D+\frac1{pq}\leq\frac3D.\qedhere
\]
\end{proof}

For a block with $D_i=D$, the lemma applies to every $ka_{i,n}$ by
taking $h=kn$. The number $Dp_iq_i/(p_iq_i-1)$ is used only to count
these multiples; it is not inserted into the configuration.

\subsection{Proof of the finite extension}

\begin{proof}[Proof of Proposition~\ref{prop:finite-extension}]
Let the initial configuration consist of $t$ blocks, with point set
$F_0$. Choose $D$ by Lemma~\ref{lem:grid}. We keep $D$ fixed while adjoining
\[
 J=\left\lfloor D/12\right\rfloor
\]
blocks. By Lemma~\ref{lem:cost}, their residue exclusions have total
density at most $3J/D\leq1/4$. We will also show that each new block
has weight greater than $1/(24D)$, so a full round compensates for the
loss of weight caused by moving to the scale $D$.
After $s$ insertions, where $0\leq s\leq J$, we require the following
local induction conditions:
\begin{enumerate}
\item the current configuration is admissible and retains all preceding
block data;
\item $D_j=D$ and $p_jq_j\geq2D$ for $t<j\leq t+s$;
\item $p_j\nmid D$ for every private prime currently present;
\item its point set $F$ satisfies
$W(F)\geq W(F_0)+s/(24D)$.
\end{enumerate}
These hold for $s=0$. Suppose they hold for $s<J$, and let $i=t+s+1$
be the next block index.

\smallskip
\noindent\emph{The private prime and the available integers.}
Choose a prime $p_i$ greater than $D$, all preceding private primes,
all the preceding integers $q_jD_jn$, and
\[
 \left(\frac1{100}-\sum_{j<i}\frac1{p_j}\right)^{-1}.
\]
Admissibility makes the quantity in parentheses positive, and all the
bounds are finite. This choice preserves the strict reciprocal bound and ensures
that $p_i$ divides neither $D$ nor any preceding $q_jD_jn$.

Define the periodic set
\begin{equation}\label{eq:round-filter}
 E=\left\{\ell\in\N:
 \begin{array}{ll}
 p_j\nmid\ell & (1\leq j\leq i),\\
 \ell\bmod p_jq_j\notin\mathcal B_{p_j,q_j}(D)
     & (t<j<i)
 \end{array}\right\}.
\end{equation}
The first row excludes multiples of all preceding private primes and
of the newly chosen $p_i$. It handles the initial blocks; the second
row handles those already inserted on the same grid.
Take the product of the displayed moduli as a common period. For each
modulus, its residue classes have equally many lifts to this period,
so the union bound and Lemma~\ref{lem:cost} give
\begin{equation}\label{eq:round-density}
 \dens(E)\geq1-\sum_{j=1}^i\frac1{p_j}
       -\sum_{j=t+1}^{i-1}
           \frac{|\mathcal B_{p_j,q_j}(D)|}{p_jq_j}
 >\frac{99}{100}-\frac{3s}{D}
 \geq\frac{37}{50}>\frac23.
\end{equation}
No coprimality of the moduli is needed, and each private prime is counted
only once. Lemmas~\ref{lem:grid} and~\ref{lem:cost}, applied respectively
to the initial blocks and those inserted in this round, give
\begin{equation}\label{eq:unperturbed-separation}
 |ka-D\ell|>1\qquad(a\in F,\ \ell\in E,\ k\in\Z).
\end{equation}

The set $E$, including its period, is now fixed. By
Lemma~\ref{thm:selection}, choose a finite primitive $P_i\subset E$
such that
\begin{equation}\label{eq:labels-choice}
 \begin{gathered}
 W(P_i)>\frac16,\qquad \min P_i\geq2D,\\
 D\min P_i>\max\{4p_i,M(F)+1\}.
 \end{gathered}
\end{equation}
The bound $n\geq2D$ will give the weight comparison at the end of the
step. The other two bounds ensure $a_{i,n}/p_i>4$ and place the new
block beyond every old point. All can be imposed through the threshold
in Lemma~\ref{thm:selection}, since $F,D,p_i$ are already fixed.
The positive weight ensures that $P_i$ is nonempty.

\smallskip
\noindent\emph{The auxiliary prime and the displacement.}
Put $U=D\max P_i$. For each $a\in F$, write $a=A_a/B_a$ in reduced
form, with $B_a>0$. Choose a prime $q_i$, different from all
$p_1,\ldots,p_i$, so large that
\begin{equation}\label{eq:auxiliary-choice}
 \begin{gathered}
 p_iq_i\geq2D,\qquad p_iq_i-1\geq2U^{13},\\
 \frac{U}{p_iq_i-1}<\frac1{B_a}\quad(a\in F).
 \end{gathered}
\end{equation}
These are finitely many lower bounds, so such a prime exists. Set
$D_i=D$ and define the block by~\eqref{eq:block-data}. For $m=Dn$ with
$n\in P_i$, its displacement is
\begin{equation}\label{eq:eta-small}
 a_{i,n}-m=\frac{m}{p_iq_i-1}
 \leq\frac{m}{2U^{13}}\leq\frac12m^{-12}.
\end{equation}
The auxiliary prime is chosen after the finite set $P_i$ precisely so
that it can meet all these bounds without changing $E$ or $P_i$.

\smallskip
\noindent\emph{Separation.}
Fix an old point $a=A_a/B_a$, a new point $a_{i,n}$, and $k\in\Z$.
With $m=Dn$, equation~\eqref{eq:unperturbed-separation} and the fact
that $ka-m\in B_a^{-1}\Z$ give the alternatives
\[
 ka-m<-1\qquad\hbox{or}\qquad ka-m\geq1+\frac1{B_a}.
\]
The displacement in~\eqref{eq:eta-small} is positive and, by
\eqref{eq:auxiliary-choice}, strictly smaller than $1/B_a$. Subtracting
it preserves a distance greater than $1$ in both cases. Thus
$|ka-a_{i,n}|>1$ for every integer $k$, with no bound on its size.

In the opposite direction,~\eqref{eq:labels-choice} gives
$a_{i,n}>M(F)+1$, so $ka_{i,n}-a>1$ for every $k\in\N$.

For distinct $n,n'\in P_i$ and $k\in\N$, primitivity gives
$kn\ne n'$. Hence, within the new block,
\[
 |ka_{i,n}-a_{i,n'}|
 =\frac{Dp_iq_i}{p_iq_i-1}|kn-n'|>D>1.
\]
This proves the required separation and ordering of the blocks.

\smallskip
\noindent\emph{Preserving the arithmetic conditions.}
For~\eqref{eq:private-invariant}, the new $p_i$ exceeds every old
$q_jD_jn$, while the preceding private primes already avoid these
integers. For a new integer $q_iDn$, every $p_j$ with $j\leq i$
avoids each of its three factors: it avoids $q_i$ by the choice of the
auxiliary prime, $D$ by induction and $p_i>D$, and $n$ by
$P_i\subset E$. Since $p_j$ is prime, $p_j\nmid q_iDn$.
Finally, $a_{i,n}>Dn>4p_i$, and~\eqref{eq:eta-small} verifies the
remaining part of~\eqref{eq:block-size}. All admissibility conditions
and the first three local induction conditions are preserved.

\smallskip
\noindent\emph{Weight and completion of the round.}
Since $1<p_iq_i/(p_iq_i-1)<2$ and $n\geq2D$, we have
$a_{i,n}<2Dn\leq n^2$, and hence $\log a_{i,n}\leq2\log n$.
Consequently
\[
 \wt(a_{i,n})\geq\frac{\wt(n)}{4D},\qquad
 W(A_i)>\frac1{24D}.
\]
The map $n\mapsto a_{i,n}$ is injective, so summing the first inequality
over $P_i$ and using~\eqref{eq:labels-choice} gives the second.
Disjointness from $F$ gives the fourth local induction condition for
$F\cup A_i$. This completes the insertion step.

Finite induction now supplies all $J$ blocks. Let $F'$ be the resulting
point set. Since $D\geq48$,
\[
 J\geq D/12-1\geq D/24,
 \qquad W(F'\setminus F_0)\geq\frac{J}{24D}\geq\frac1{576}.
\]
The construction has changed none of the preceding block data. Its
final admissibility allows Lemma~\ref{lem:grid} to be applied again
at the start of the next round, using every integer $q_iD_i n$ then
present. The new grid need not equal $D$; it affects only future blocks.
\end{proof}

\section{Primitive sets in periodic sets}
\label{sec:analytic}
We prove Lemma~\ref{thm:selection}. For $n\in\N$, let $\Om(n)$ count
prime factors with multiplicity, with $\Om(1)=0$. Integers with the same
positive value of $\Om$ form a primitive set: if $n$ properly divides
$n'$, then $\Om(n')>\Om(n)$. We seek arbitrarily large levels whose
weight in $E$ exceeds $1/6$. For $0<u<1$, we average the weights of the levels
$\Om(n)=r$, $r\geq0$, using the coefficients $(1-u)u^r$, whose sum is $1$,
and then let $u$ tend to $1$ from below.

For a fixed modulus $L\geq1$, residue $b\pmod L$ and $0<u<1$, set
\[
 A_{L,b}(u)=\sum_{\substack{n\geq2\\n\equiv b\pmod L}}
       \frac{u^{\Om(n)}}{n\log n}.
\]
We first justify convergence and prove
\begin{equation}\label{eq:residue-abel}
 \liminf_{u\uparrow1}(1-u)A_{L,b}(u)\geq\frac1{4L}.
\end{equation}
Summing this estimate over the allowed residue classes of $E$ will
give a lower bound greater than $1/6$ for the normalized average.
We then extract a finite set from a sufficiently high level.
All limits are taken with $L$ fixed.

\subsection{From logarithmic weights to Euler products}
Dirichlet characters will impose the congruence condition for coprime
residues. To obtain the weight $1/(n\log n)$ from an Euler product, we
use the identity
\[
 \frac1{n\log n}=\int_0^\infty n^{-1-t}\,dt\qquad(n\geq2).
\]
For a Dirichlet character $\chi$ modulo $L$ and $t>0$, define
\[
 G_\chi(u,t)=\prod_{p\text{ prime}}(1-u\chi(p)p^{-1-t})^{-1}
       =\sum_{n\geq1}\frac{u^{\Om(n)}\chi(n)}{n^{1+t}}.
\]
These products and series converge absolutely. Expanding the logarithms
of their Euler factors gives
\begin{equation}\label{eq:log-product}
 \begin{aligned}
 \log|G_\chi(u,t)|&=u\log|L(1+t,\chi)|+\Re R_\chi(u,t),\\
 R_\chi(u,t)&=\sum_p\sum_{j\geq2}
       \frac{(u^j-u)\chi(p)^j}{j p^{j(1+t)}}.
 \end{aligned}
\end{equation}
Here logarithms of absolute values are real logarithms; the equality
follows by taking real parts of the absolutely convergent logarithmic
Euler series. Since $0\leq u-u^j\leq(j-1)(1-u)$,
\begin{equation}\label{eq:log-remainder}
 |R_\chi(u,t)|\leq(1-u)\sum_{n\geq2}\sum_{j\geq2}n^{-j}
   =(1-u)\sum_{n\geq2}\frac1{n(n-1)}=1-u.
\end{equation}
This is uniform in the character and $t>0$.

For a nonprincipal character, $L(s,\chi)$ is continuous on $[1,2]$
and hence bounded there. Equations~\eqref{eq:log-product}--\eqref{eq:log-remainder}
therefore bound $|G_\chi(u,t)|$ uniformly for $0<u<1$ and $0<t\leq1$.
No non-vanishing assertion at $s=1$ is used. For the principal character
$\chi_0$, put $\kappa_L=\varphi(L)/L$, with $\varphi(1)=1$.
The identity
\[
 L(1+t,\chi_0)=\zeta(1+t)\prod_{p\mid L}(1-p^{-1-t})
\]
and $t\zeta(1+t)\to1$ show that
$tL(1+t,\chi_0)\to\kappa_L$. Thus, for some fixed
$\varepsilon\in(0,1]$ and for $1/2\leq u<1$,
\begin{equation}\label{eq:principal-lower}
 G_{\chi_0}(u,t)\geq\frac12
       \left(\frac{\kappa_L}{2t}\right)^u
       \qquad(0<t\leq\varepsilon).
\end{equation}
Indeed, $L(1+t,\chi_0)\geq\kappa_L/(2t)$ there, and
$e^{\Re R}\geq e^{-1/2}>1/2$.
These standard properties of Dirichlet $L$-functions follow from their
Euler products and the bounded partial sums of nonprincipal characters;
see~\cite[Sections~2 and~6]{Tao2014}.

Let $G_1$ denote the unrestricted product (modulus one).
For some fixed $\varepsilon_0\in(0,1]$ and $0<t\leq\varepsilon_0$,
\eqref{eq:log-product} gives
$G_1(u,t)\leq e\zeta(1+t)^u\leq2e\,t^{-u}$.
For every character, the Dirichlet series also gives
\[
 |G_\chi(u,t)-1|\leq\sum_{n\geq2}n^{-1-t}\ll2^{-t}
       \qquad(t\geq1),
\]
uniformly in $u$. Continuity handles the intervening compact interval.
Thus $G_1(u,t)-1$ is integrable on $(0,\infty)$ for each fixed $u<1$.
Tonelli's theorem for its nonnegative series and
$|\chi(n)|\leq1$ justify termwise integration:
\begin{equation}\label{eq:weighted-integral}
 J_\chi(u):=\int_0^\infty(G_\chi(u,t)-1)\,dt
      =\sum_{n\geq2}\frac{u^{\Om(n)}\chi(n)}{n\log n},
\end{equation}
with absolute convergence.

\subsection{The principal character and the residue classes}
If $b$ is invertible modulo $L$, character orthogonality gives
\[
 A_{L,b}(u)=\frac1{\varphi(L)}
       \sum_{\chi\bmod L}\overline{\chi(b)}J_\chi(u).
\]
For nonprincipal $\chi$, the preceding estimates make $J_\chi(u)$
bounded uniformly in $u$. For $\chi_0$, the Dirichlet series shows that
the integrand is nonnegative, so integrating
\eqref{eq:principal-lower} over $(0,\varepsilon)$ gives
\[
 J_{\chi_0}(u)\geq
 \frac{(\kappa_L/2)^u\varepsilon^{1-u}}{2(1-u)}-\varepsilon.
\]
Consequently
\begin{equation}\label{eq:character-limits}
 (1-u)J_\chi(u)\to0\quad(\chi\ne\chi_0),\qquad
 \liminf_{u\uparrow1}(1-u)J_{\chi_0}(u)\geq\frac{\kappa_L}{4}.
\end{equation}

In the orthogonality identity, the principal contribution therefore
has normalized lower limit at least
$\kappa_L/(4\varphi(L))=1/(4L)$, while every other normalized term
tends to zero. Taking real parts proves~\eqref{eq:residue-abel}
for coprime residues, including $L=1$.

It remains to treat residues that are not coprime to the modulus.
Writing $n=dm$ reduces the congruence to a coprime one; the resulting
change from $\log m$ to $\log(dm)$ has bounded total cost.
More precisely, choose $0\leq b<L$ and put
$d=\gcd(b,L)$, $M=L/d$ and $c=b/d$. The residue $c$ is invertible
modulo $M$, including the case $M=1$. For $m\geq2$,
\[
 0\leq\frac1{m\log m}-\frac1{m\log(dm)}
       \leq\frac{\log d}{m(\log m)^2}.
\]
The series $C_2=\sum_{m\geq2}1/(m(\log m)^2)$ converges by the
integral test. Using $\Om(dm)=\Om(d)+\Om(m)$ and $u^{\Om(m)}\leq1$
therefore gives
\[
 A_{L,b}(u)\geq\frac{u^{\Om(d)}}d A_{M,c}(u)
                        -\frac{\log d}{d}C_2.
\]
The possible omitted term $m=1$, with $d\geq2$, is nonnegative.
After multiplication by $1-u$, the error tends to zero and the leading
coefficient tends to $1/d$. The coprime case gives a lower limit of at
least $1/(4dM)=1/(4L)$, proving~\eqref{eq:residue-abel} for every residue.

\subsection{Extracting a finite primitive set}
\begin{proof}[Proof of Lemma~\ref{thm:selection}]
Fix a period $L$ of $E$. Summing~\eqref{eq:residue-abel} over its
allowed residue classes gives
\[
 \liminf_{u\uparrow1}(1-u)
       \sum_{\substack{n\geq2\\n\in E}}u^{\Om(n)}\wt(n)
       \geq\frac{\dens(E)}4>\frac16.
\]
For each $r\geq0$, let $f_r$ be the weight of the corresponding level in $E$:
\[
 f_r=\sum_{\substack{n\geq2,\ n\in E\\\Om(n)=r}}\wt(n).
\]
Every $f_r$ is finite, since the convergent series at $u=1/2$
majorizes $2^{-r}f_r$. Regrouping the nonnegative series gives
\[
 \sum_{\substack{n\geq2\\n\in E}}u^{\Om(n)}\wt(n)
       =\sum_{r\geq0}u^rf_r.
\]
If $f_r\leq1/6$ for all $r\geq R$, then
\[
 (1-u)\sum_{r\geq0}u^rf_r
       \leq\frac16+(1-u)\sum_{r<R}f_r,
\]
contradicting the preceding lower limit. Hence $f_r>1/6$ for
arbitrarily large $r$. Choose one with $r\geq N\geq2$.
Since $f_r>1/6$, there is a finite set
$P\subset\{n\in E:\Om(n)=r\}$ with $W(P)>1/6$.
Every element of $P$ is at least $2^r\geq N$, and the common positive
value of $\Om$ makes $P$ primitive. This proves the lemma.
\end{proof}
